class C4MovableWnd { constructor(id) { this.id = id; this.wnd = document.getElementById(id); if (!this.wnd) { this.createWnd(); } this.setupEvents(); } createWnd() { this.wnd = document.createElement('div'); this.wnd.id = this.id; this.wnd.style.position = 'absolute'; this.wnd.style.width = '300px'; this.wnd.style.height = '200px'; this.wnd.style.backgroundColor = '#fff'; this.wnd.style.border = '1px solid #ccc'; this.wnd.style.boxShadow = '0 0 10px rgba(0, 0, 0, 0.1)'; this.wnd.style.zIndex = '1000'; const titleBar = document.createElement('div'); titleBar.style.height = '30px'; titleBar.style.backgroundColor = '#f0f0f0'; titleBar.style.cursor = 'move'; titleBar.textContent = 'Title'; this.wnd.appendChild(titleBar); document.body.appendChild(this.wnd); } setupEvents() { let isDragging = false; let offsetX, offsetY; const titleBar = this.wnd.firstChild; titleBar.addEventListener('mousedown', (e) => { isDragging = true; offsetX = e.clientX - this.wnd.offsetLeft; offsetY = e.clientY - this.wnd.offsetTop; }); document.addEventListener('mousemove', (e) => { if (isDragging) { this.wnd.style.left = (e.clientX - offsetX) + 'px'; this.wnd.style.top = (e.clientY - offsetY) + 'px'; } }); document.addEventListener('mouseup', () => { isDragging = false; }); // 处理移动端触摸事件 titleBar.addEventListener('touchstart', (e) => { isDragging = true; const touch = e.touches[0]; offsetX = touch.clientX - this.wnd.offsetLeft; offsetY = touch.clientY - this.wnd.offsetTop; e.preventDefault(); // 防止背景网页滚动 }); document.addEventListener('touchmove', (e) => { if (isDragging) { const touch = e.touches[0]; this.wnd.style.left = (touch.clientX - offsetX) + 'px'; this.wnd.style.top = (touch.clientY - offsetY) + 'px'; e.preventDefault(); // 防止背景网页滚动 } }); document.addEventListener('touchend', () => { isDragging = false; }); } toggleWnd() { if (this.wnd.style.display === 'none') { this.wnd.style.display = 'block'; } else { this.wnd.style.display = 'none'; } } } const o = new C4MovableWnd（'id4-div-ui'）; o.toggleWnd();
